static func _date_str_to_jdn ( date_str : String ) -> int :
var p : PackedStringArray = date_str . split ( "-" )
if p . size ( ) != 3 :
return 0
var y : int = int ( p [ 0 ] )
var m : int = int ( p [ 1 ] )
var d : int = int ( p [ 2 ] )
var a : int = int ( ( 14 - m ) / 12 )
var yy : int = y + 4800 - a
var mm : int = m + 12 * a - 3
return d + int ( ( 153 * mm + 2 ) / 5 ) + 365 * yy + int ( yy / 4 ) - int ( yy / 100 ) + int ( yy / 400 ) - 32045
